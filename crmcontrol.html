SQL sugerido para Supabase create table if not exists public.eventos ( id uuid primary key default gen_random_uuid(), cliente text not null, agasajado text, dni text, telefono text, tipo_evento text default 'Cumpleaños', edad integer, invitados integer, fecha_evento date not null, hora_inicio text, hora_fin text, lugar text, direccion text, paquete text, detalle_servicio text, adicionales text, observaciones text, movilidad text default 'Incluida', estado_pago text default 'Reservado', total numeric default 0, adelanto numeric default 0, saldo numeric default 0, calendar_event_id text, drive_pdf_url text, created_at timestamptz default now(), updated_at timestamptz default now(), juegos_elegidos text ); create extension if not exists pgcrypto; alter table public.eventos enable row level security; create policy "allow all for anon" on public.eventos for all using (true) with check (true);
